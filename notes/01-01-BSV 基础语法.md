# 基础语法与框架

包名与文件名相同，习惯驼峰，模块名习惯`mk`作为前缀标识度++，标识符大小写敏感，类型/常量使用大驼峰，变量/函数习惯小驼峰

```bluespec
// 基础类型：
Bit#(233) emm = 233'd666;   // 233位bit，值为666
Int#( 64) xxx = -5555555;   // 64位有符号整数
UInt#(16) yyy =        5;   // 16位无符号整数
Bool     flag = True || False;

int         i =       32;   // Int#(32)的别名

let x = 42;                 // let 开启类型推导，默认整数 Integer
let y = 8'o7;               // y 是 Bit#(8)
let z <- fifo.get();        // 推断为 ActionValue 返回类型

// 用户类型：
typedef enum {    // 枚举
    IDLE,
    PEDDING,
    RUNNING,
    DONE,
    ERROR
} State deriving (Bits, Eq, FShow);  // deriving 类似于 rust 派生宏

typedef struct {  // 结构体
    Bit#( 8) addr;
    Bit#(32) data;
    Bool    ready;
    Bool    valid;
} Entry deriving (Bits, Eq);

typedef Bit#(16) Pixel;     // 类型别名
typedef Vector#(64, Pixel) LineBuf;

// 可选值（Maybe 属 Prelude 内置，无需 import）
Maybe#(int) maybeValue = tagged Valid 42;
Maybe#(Bool) xxx = tagged Invalid;

let valid = isValid(maybeValue);        // 变量别与内建 isValid 同名（否则自遮蔽 P0040）
let value = fromMaybe(0, maybeValue);   // 注意 default 是保留字，用具体默认值

case (maybeValue) matches
    tagged Valid .val: $display("Value is %d", val);
    tagged Invalid: $display("Value is None");
endcase

// 向量
import Vector::*;
import BuildVector::*;                        // vec(...) 字面量在 BuildVector 包，不在 Vector

Vector#(4, int) list = newVector;             // 未初始化
Vector#(4, Bool) bitmap = replicate(True);    // 全部填充True
Vector#(4, int) lit = vec(1, 2, 233, 4);      // 字面量列表（需 import BuildVector）

// —— 区分四种「赋值/写」——
Int#(32) item = list[0];                            // 读：索引取元素（=）
Vector#(4, int) list2 = update(list, 1, 233);       // 纯值向量「写」= 函数式 update（不可变，返回新向量）
Reg#(int) r <- mkReg(0);  r <= 88;                  // 实例化赋予给变量：<-，寄存器写：<=
Vector#(4, Reg#(int)) rv <- replicateM(mkReg(0));  rv[1] <= 233;   // 寄存器向量：元素才能 <=

Vector#(4, int) mapped = map(f, v);            // 映射
Vector#(8, int) concated = append(v1, v2);     // 拼接
int sum = fold(add, v);                        // 折叠

// 元组（Tuple 属 Prelude 内置，无需 import）
Tuple2#(int, Bool) t2 = tuple2(42, True);      // 二元组（类型名首字母大写 Bool，不是 bool）
match {.a, .b} = t2;    // a=42, b=True

Tuple3 ...   // 三元组就免了，访问：tpl_1(t2), tpl_2(t2) 访问第一个和第二个元素...

// 寄存器
Reg#(Bit#(8)) counter <- mkReg(66);      // 带初值的初始化寄存器
let x = counter;                         // 等价于 counter._read()
counter <= 88;                           // 等价于 counter._write(88)
counter <= counter + 1;                  // 原子操作，读旧值，加1，写回
mkReg(x)                                 // 初始值x，读<写（读写同时出现在表达式，则读的是旧值）
mkRegU()                                 // 未初始化，用于临时存储
mkCReg(n)                                // n端口并发寄存器，用于高并发
mkDWire(x)                               // 组合线，写<读（先写，读的都是新写入的值）

// 规则 - 无条件则永远触发，带条件则条件满足才触发，原子时序
rule rl_always;
    counter <= counter + 1;
endrule

rule rl_process (state == RUNNING);
    data <= process(data);
    reg1 <= reg1 + 1;
    reg2 <= reg1;      // 多个动作同时执行，读的是reg1旧值
    fifo.enq(data);
    if (reg1 > 10)
        state <= DONE;
    else
        state <= RUNNING;
endrule

// 规则的优先级控制
(* descending_urgency = "rl_high, rl_low" *)  // 声明规则优先级
...
(* mutually_exclusive = "rl_a, rl_b" *)  // 声明俩规则永远互斥
rule rl_a; ... endrule
rule rl_b; ... endrule
(* execution_order = "rl_first, rl_second" *)   // 执行顺序
...

// 接口
interface MyIFC;
    method Int#(32)          get();           // method t               name() 无副作用 返回值类型t
    method Action            add(int delta);  // method Action          name() 有副作用 无返回值
    method ActionValue#(int) cls();           // method ActionValue#(t) name() 有副作用 返回值类型t
endinterface

interface GenIFC #(type T);          // 泛型/多类型/参数化接口
    method T      get();
    method Action enq(T x);
endinterface

interface GenIFC #(type T, type F);  // 嵌套接口
    interface Put#(req_t) req;
    interface Get#(rsp_t) rsp;
endinterface

// 方法实现（模块须实现接口声明的【所有】方法，不能多也不能少；这里省略 get/cls 仅作示意）
module mkCounter(MyIFC);
    Reg#(int) count <- mkReg(0);

    method Bool isZero();                          // 值方法：返回类型在前
        return (count == 0);
    endmethod

    method Action add(int delta) if (count != 0); // 带隐式条件(implicit condition)：count==0 时本方法不就绪
        count <= count + delta;
    endmethod

    // 其余方法 get/cls 实现 ... 读到的是旧值，而非本拍新写入的值
endmodule

// 系统函数
$display("Value: %d, Hex: %h, Bin: %b", x, x, x);  $write("format\n", args...);
$finish; $finish(0); $finish(1); $stop;    // 结束仿真/结束附带返回值/暂停仿真进入交互模式
$dumpfile("waves.vcd"); $dumpvars;  // 指定波形文件，转储所有变量
$time; $stime; $realtime; #10;   // 当前仿真时间/32位时间/实数时间/等待10个时间单位（仅用于initial）

/*
   多行注释
*/
```

**基础TestBench示例**：

```bluespec
// 最简TB
package Testbench;

module mkTestbench (Empty); // Empty-空接口，无任何方法

  rule rl_run;
    $display("Hello, World!");
    $finish;
  endrule

endmodule

endpackage

// 完整示例
package Testbench;

// 导入需要的包
import StmtFSM::*;
import Vector::*;
import MyDesign::*;     // 待测设计

// 类型定义
typedef enum {
  RESET,
  RUNNING,
  VERIFY,
  DONE
} TestState deriving (Bits, Eq, FShow);

// 顶层模块
module mkTestbench (Empty);

  // ========== 状态元素 ==========
  Reg#(TestState) state <- mkReg(RESET);
  Reg#(Bit#(32)) cycle <- mkReg(0);
  Reg#(int) test_num <- mkReg(0);

  // ========== DUT 实例化 ==========
  MyIFC dut <- mkMyDesign;

  // ========== 辅助函数 ==========
  function Action check_equal(int expected, int actual);
    action
      if (expected == actual)
        $display("[OK] Cycle %0d: expected=%d, got=%d", cycle, expected, actual);
      else
        $display("[FAIL] Cycle %0d: expected=%d, got=%d", cycle, expected, actual);
    endaction
  endfunction

  // ========== 规则 ==========

  // 周期计数
  rule rl_cycle;
    cycle <= cycle + 1;
  endrule

  // 状态机
  rule rl_reset (state == RESET);
    $display("Cycle %0d: Starting test", cycle);
    dut.reset();
    state <= RUNNING;
  endrule

  rule rl_running (state == RUNNING);
    if (test_num < 100) begin
      // 生成激励
      let stimulus = test_num * 2;
      dut.send(stimulus);
      $display("Cycle %0d: Sent %d", cycle, stimulus);
      test_num <= test_num + 1;
    end
    else begin
      state <= VERIFY;
    end
  endrule

  rule rl_verify (state == VERIFY && dut.done());
    let result <- dut.getResult();
    check_equal(9900, result);  // 0+2+4+...+198 = 9900
    state <= DONE;
  endrule

  rule rl_done (state == DONE);
    $display("Cycle %0d: Test completed", cycle);
    $finish;
  endrule

  // 超时保护
  rule rl_timeout (cycle > 10000);
    $display("ERROR: Timeout at cycle %0d", cycle);
    $finish(1);
  endrule

endmodule

endpackage
```

**其他补充**：

Reg#(int) temp <- mkRegU(); 未初始化 <=> 你向编译器承诺，在读取这个寄存器的值之前，一定会先写入它

tagged 是操作联合体的关键字

```bluespec
// 定义一个名为 MaybeInt 的联合体类型
typedef union tagged {
    void Invalid;      // 情况1：无效，不携带数据
    int  Valid;        // 情况2：有效，携带一个 int 类型的数据
} MaybeInt deriving (Bits, Eq);

// 创建 tagged 值
MaybeInt no_value = tagged Invalid;
MaybeInt some_value = tagged Valid 42;

// 使用 case 语句匹配 tag，安全地取出数据
MaybeInt x = ...;
case (x) matches
    tagged Invalid : $display("没有值");
    tagged Valid .v : $display("值是：%d", v);
endcase

// 带有结构体的联合体
typedef union tagged {
    void DivByZero;
    struct { int quotient; int remainder; } DivResult;
} ArithResult;

// 创建一个结构体 tag 的值
ArithResult res = tagged DivResult { quotient: 5, remainder: 2 };

// 实际的 Maybe 定义类似这样，即：多态联合体
typedef union tagged {
    void Invalid;
    a Valid;
} Maybe#(type a);
```

deriving 让编译器自动为你的自定义类型实现一些常用且“显而易见”的接口（类型类）


rule 是时序逻辑，有一个可选的守卫条件 function 是组合逻辑

**关于接口的另一个角度**：

| 方法类型 | 语法 | 综合行为 |
|:---:|:---:|:---:|
| `Action` | `method Action m(args);` | 有 EN 信号，有 RDY |
| `value method` | `method t m(args);` | 有 RDY 信号，组合逻辑输出 |
| `ActionValue#(t)` | `method ActionValue#(t) m(args);` | 既有 EN 又有 RDY，返回数据 |

> `mkReg`的`_read < _write` -> 读写同时则读出旧值 -> 速记：`write`在`read`“之后”
