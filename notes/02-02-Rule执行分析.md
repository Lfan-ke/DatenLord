# 分析Rule编译后的结果

## 简单的Reg写入

> 对应文件`execs/03-01-01-SimpleReg/SimpleRegConflict.bsv`
>
> cd execs/03-01-01-SimpleReg
>
> bsc -verilog -show-schedule -g mkSimpleRegConflict SimpleRegConflict.bsv

简单的两个`Rule`分别向两个`Reg`写入不同的值，`value`方法用来获取写入的值。

```bluespec
package SimpleRegConflict;

interface TestIFC;
    method Bit#(8) value;
endinterface

(* synthesize *)
module mkSimpleRegConflict(TestIFC);

    Reg#(Bit#(8)) r <- mkReg(0);  // r < w

    rule writeA;
        r <= 8'hAA;
    endrule

    rule writeB;
        r <= 8'h55;
    endrule

    method Bit#(8) value;
        return r;
    endmethod

endmodule

endpackage
```

可以看到有两个Rule，同时向`r`写了值，调度顺序没有注解，默认先A后B：

```sched

//
// On Thu May 21 08:15:14 CST 2026

// Ports:

// CLK                            I     1 clock
// RST_N                          I     1 reset
//

//

`else
  `define BSV_ASSIGNMENT_DELAY
`endif

`ifdef BSV_POSITIVE_RESET
  `define BSV_RESET_VALUE 1'b1
  `define BSV_RESET_EDGE posedge
`else
  `define BSV_RESET_VALUE 1'b0
  `define BSV_RESET_EDGE negedge
`endif

module mkSimpleRegConflict(CLK,
			   RST_N,

			   value,
			   RDY_value);
  input  CLK;
  input  RST_N;

  // value method value
  output [7 : 0] value;
  output RDY_value;

  // signals for module outputs
  wire [7 : 0] value;
  wire RDY_value;

  // register r
  reg [7 : 0] r;
  wire [7 : 0] r$D_IN;
  wire r$EN;

  // value method value
  assign value = r ;
  assign RDY_value = 1'd1 ;

  // register r
  assign r$D_IN = 8'h55 ;  // 可以看到只剩余 RuleB 的 8'h55 了
  assign r$EN = 1'b1 ;

  // handling of inlined registers

  always@(posedge CLK)
  begin
    if (RST_N == `BSV_RESET_VALUE)
      begin
        r <= `BSV_ASSIGNMENT_DELAY 8'd0;
      end
    else
      begin
        if (r$EN) r <= `BSV_ASSIGNMENT_DELAY r$D_IN;
      end
  end

  // synopsys translate_off
  `ifdef BSV_NO_INITIAL_BLOCKS
  `else // not BSV_NO_INITIAL_BLOCKS
  initial
  begin
    r = 8'hAA;
  end
  `endif // BSV_NO_INITIAL_BLOCKS
  // synopsys translate_on
endmodule  // mkSimpleRegConflict
```

## 有冲突的场景

一个PipelineFIFO：

```bluespec
package PipelineFIFO;

import FIFOF::*;

interface TestIFC;
    method Action enq(Bit#(8) x);
    method Action deq;
    method Bit#(8) first;
    method Bool notEmpty;
    method Bool notFull;
endinterface

(* synthesize *)
module mkPipelineFIFO(TestIFC);

    FIFOF#(Bit#(8)) fifo <- mkFIFOF;

    method Action enq(Bit#(8) x) if (fifo.notFull);
        fifo.enq(x);
    endmethod

    method Action deq if (fifo.notEmpty);
        fifo.deq;
    endmethod

    method Bit#(8) first if (fifo.notEmpty);
        return fifo.first;
    endmethod

    method Bool notEmpty;
        return fifo.notEmpty;
    endmethod

    method Bool notFull;
        return fifo.notFull;
    endmethod

endmodule

endpackage
```
